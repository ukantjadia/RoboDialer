# Map the request Origin to a CORS-approved value

map $http_origin $cors_origin {

    default "";

    ~^https://(www\.)?saasquatchleads\.com$ $http_origin;

    ~^https://app\.saasquatchleads\.com$ $http_origin;

    ~^https://sandboxdev\.saasquatchleads\.com$ $http_origin;

    ~^https://main\.d2fzqm2i2qb7f3\.amplifyapp\.com$ $http_origin;

    ~^http://localhost:3000$ $http_origin;

    ~^https://sandbox-api\.saasquatchleads\.com$ $http_origin;

}
